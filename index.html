// ==UserScript== // @name FREEDOM-INSIDE - Maps Lead Extractor & WA Blaster // @namespace http://tampermonkey.net/ // @version 1.0 // @description Ekstrak leads Google Maps & WA Blaster otomatis untuk FREEDOM-INSIDE // @author Puguh // @match https://www.google.com/maps/* // @grant none // == ==/UserScript== (function() { 'use strict'; // --- GANTI DENGAN URL WEB APP GOOGLE APPS SCRIPT ANDA --- const WEB_APP_URL = "https://script.google.com/macros/s/AKfycbwkOanXlqtH-P7mP8TG8TYwSYhSSBJ7GPqMwthojXZ-g-1Zthy3ozMoS6yUQ89IDueZjA/exec"; // --------------------------------------------------------- let extractedData = []; // Buat Tombol Melayang di Google Maps function createFloatingButton() { if (document.getElementById('freedom-floating-panel')) return; const panel = document.createElement('div'); panel.id = 'freedom-floating-panel'; panel.innerHTML = `
⚡ FREEDOM-INSIDE
Status: Siap Ekstrak
`; document.body.appendChild(panel); document.getElementById('btn-extract').addEventListener('click', startExtraction); document.getElementById('btn-blast').addEventListener('click', openBlastModal); } // Fungsi Otomatis Scroll & Ambil Data Google Maps async function startExtraction() { const statusEl = document.getElementById('freedom-status'); const btnExtract = document.getElementById('btn-extract'); btnExtract.disabled = true; statusEl.innerText = "Status: Sedang menggulir & menyedot data..."; extractedData = []; // Cari elemen panel daftar tempat di Google Maps (biasanya bersilangan dengan role feed) let feed = document.querySelector('div[role="feed"]'); if (!feed) { alert("Harap lakukan pencarian di Google Maps terlebih dahulu (contoh: 'Cafe di Yogyakarta') hingga daftar tempat muncul di sebelah kiri!"); btnExtract.disabled = false; statusEl.innerText = "Status: Gagal (Buka daftar tempat dulu)"; return; } // Auto-scroll ke bawah untuk memuat data sebanyak-banyaknya let lastHeight = feed.scrollHeight; for (let i = 0; i < 10; i++) { // Maksimal 10 kali scroll otomatis feed.scrollTo(0, feed.scrollHeight); await new Promise(resolve => setTimeout(resolve, 1500)); if (feed.scrollHeight === lastHeight) break; lastHeight = feed.scrollHeight; } // Ambil elemen kartu bisnis let items = document.querySelectorAll('div[role="feed"] > div > div[jsaction]'); items.forEach(item => { let nameEl = item.querySelector('.fontHeadlineSmall'); let phoneEl = item.querySelector('span[aria-label*="Telepon"], span.USccKb'); // Deteksi nomor telepon let name = nameEl ? nameEl.innerText.trim() : "Tanpa Nama"; let phone = ""; // Cari teks yang mirip nomor telepon di dalam kartu let textContent = item.innerText; let phoneMatch = textContent.match(/(\+62|62|0)[0-9\- ]{8,15}/); if (phoneMatch) { phone = phoneMatch[0].replace(/[\-\s]/g, ''); if (phone.startsWith('0')) phone = '62' + phone.substring(1); } if (name !== "Tanpa Nama" && phone) { // Hindari duplikat if (!extractedData.some(d => d.phone === phone)) { extractedData.push({ name: name, phone: phone }); } } }); statusEl.innerText = `Berhasil! Ditemukan ${extractedData.length} leads valid.`; btnExtract.disabled = false; if (extractedData.length > 0) { document.getElementById('btn-blast').style.display = 'block'; syncToCloud(extractedData); } else { alert("Nomor WhatsApp/Telepon tidak ditemukan pada halaman ini. Pastikan tampilan Google Maps sudah memuat informasi kontak."); } } // Kirim Data Otomatis ke Backend Google Sheets function syncToCloud(leads) { const userEmail = localStorage.getItem('user_email') || 'tamu@freedom.com'; fetch(WEB_APP_URL, { method: 'POST', body: JSON.stringify({ action: "saveLeads", email: userEmail, leads: leads }) }) .then(res => res.json()) .then(data => console.log("Cloud Sync:", data)) .catch(err => console.error("Gagal sinkron cloud:", err)); } // Tampilkan Panel WA Blaster Sederhana function openBlastModal() { let msg = prompt("Masukkan template pesan WhatsApp untuk dikirim ke leads:", "Halo Kak, salam kenal dari FREEDOM-INSIDE. Kami lihat bisnisnya sangat menarik..."); if (!msg) return; let index = 0; function sendNext() { if (index >= extractedData.length) { alert("Semua antrean WhatsApp Blaster selesai!"); return; } let lead = extractedData[index]; index++; let waUrl = `https://api.whatsapp.com/send?phone=${lead.phone}&text=${encodeURIComponent(msg.replace('{nama}', lead.name))}`; window.open(waUrl, '_blank'); // Jeda 5 detik otomatis sebelum lanjut ke kontak berikutnya agar aman setTimeout(sendNext, 5000); } sendNext(); } // Jalankan pembuat tombol saat halaman Google Maps siap window.addEventListener('load', () => { setTimeout(createFloatingButton, 3000); }); // Deteksi jika halaman berubah (SPA Google Maps) let lastUrl = location.href; new MutationObserver(() => { if (location.href !== lastUrl) { lastUrl = location.href; setTimeout(createFloatingButton, 2000); } }).observe(document, {subtree: true, childList: true}); })();
