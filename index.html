// ==UserScript== // @name FREEDOM-INSIDE - Maps Lead Extractor & WA Blaster // @namespace http://tampermonkey.net/ // @version 1.2 // @description Ekstrak leads Google Maps & WA Blaster otomatis untuk FREEDOM-INSIDE // @author Puguh // @match https://www.google.com/maps/* // @grant none // @run-at document-idle // ==/UserScript== (function () { 'use strict'; /* ========================================================= KONFIGURASI ========================================================= */ const WEB_APP_URL = "https://script.google.com/macros/s/AKfycbzMr60gyUyRa7d31dnpIM0MrGLcTk7SxFP-b0oj4M0e2p1rqu_ICH0oU-Yj9_DDwRN7bA/exec"; const panel = document.createElement('div'); panel.id = CONFIG.PANEL_ID; panel.innerHTML = `
⚡ FREEDOM-INSIDE
Status: Siap Ekstrak
Leads: 0
`; document.body.appendChild(panel); document.getElementById('freedom-close').addEventListener('click', () => panel.remove()); document.getElementById('btn-extract').addEventListener('click', startExtraction); document.getElementById('btn-blast').addEventListener('click', openBlastModal); document.getElementById('btn-stop').addEventListener('click', stopBlaster); } const updateCount = () => { const el = document.getElementById('freedom-count'); if (el) el.textContent = `Leads: ${extractedData.length}`; }; /* ========================================================= EKSTRAKSI ========================================================= */ function findFeed() { return ( document.querySelector('div[role="feed"]') || document.querySelector('div[role="main"] div[role="feed"]') ); } async function autoScrollFeed(feed) { let lastHeight = 0; let stagnant = 0; for (let i = 0; i < CONFIG.MAX_SCROLLS; i++) { feed.scrollTo(0, feed.scrollHeight); await sleep(CONFIG.SCROLL_DELAY); const newHeight = feed.scrollHeight; const endMarker = feed.querySelector('span.HlvSq'); if (newHeight === lastHeight) { stagnant++; if (stagnant >= CONFIG.STAGNANT_LIMIT) break; } else { stagnant = 0; } lastHeight = newHeight; setStatus(`Menggulir... ${extractedData.length} leads valid.`); if (extractedData.length > 0) { btnBlast.style.display = 'block'; await syncToCloud(extractedData); } else { alert('Nomor WhatsApp/Telepon tidak ditemukan.\n' + 'Pastikan kartu tempat memuat info kontak (klik tiap tempat untuk memverifikasi).'); } } catch (err) { console.error('[FREEDOM] Ekstraksi gagal:', err); setStatus('Error: ' + err.message); } finally { isExtracting = false; btnExtract.disabled = false; btnExtract.textContent = '📥 Mulai Ekstrak Data'; } } /* ========================================================= SINKRONISASI CLOUD ========================================================= */ async function syncToCloud(leads) { const userEmail = localStorage.getItem(CONFIG.STORAGE_EMAIL_KEY) || CONFIG.DEFAULT_EMAIL; try { const res = await fetch(WEB_APP_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: JSON.stringify({ action: 'saveLeads', email: userEmail, leads: leads, }), redirect: 'follow', }); if (!res.ok) throw new Error('HTTP ' + res.status); const data = await res.json().catch(() => ({})); console.log('[FREEDOM] Cloud Sync OK:', data); setStatus(`Tersimpan di cloud (${leads.length} leads).`); } catch (err) { console.error('[FREEDOM] Gagal sinkron cloud:', err); setStatus('Gagal sinkron cloud (data tetap lokal).'); } } /* ========================================================= WA BLASTER ========================================================= */ function stopBlaster() { if (blastTimer) { clearTimeout(blastTimer); blastTimer = null; } const btnStop = document.getElementById('btn-stop'); if (btnStop) btnStop.style.display = 'none'; setStatus('Blaster dihentikan.'); } function openBlastModal() { if (!extractedData.length) { alert('Tidak ada leads untuk diblast.'); return; } if (blastTimer) { alert('Blaster masih berjalan. blastMessage = msg; blastIndex = 0; const btnStop = document.getElementById('btn-stop'); btnStop.style.display = 'block'; setStatus(`Blaster aktif: 0/${total}`); const sendNext = () => { if (blastIndex >= total) { setStatus('Blaster selesai!'); btnStop.style.display = 'none'; blastTimer = null; alert('Semua antrean WhatsApp Blaster selesai!'); return; } const lead = extractedData[blastIndex++]; const text = blastMessage.replace(/\{nama\}/g, lead.name); const waUrl = `https://api.whatsapp.com/send?phone=${lead.phone}&text=${encodeURIComponent(text)}`; const win = window.open(waUrl, '_blank'); if (!win) { alert('Popup diblokir!\nIzinkan popup untuk google.com lalu ulangi.'); }, 1500); } }); observer.observe(document.body || document.documentElement, { subtree: true, childList: true, }); } /* ========================================================= BOOT ========================================================= */ window.addEventListener('load', () => { setTimeout(mountWhenReady, 2000); setupObserver(); }); })();
